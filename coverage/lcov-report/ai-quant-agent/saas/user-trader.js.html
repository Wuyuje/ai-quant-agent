function bscRpc(method, params = []) {
  return new Promise((resolve, reject) => {
    const body = JSON.stringify({ jsonrpc: '2.0', method, params, id: Date.now() });
    const url = new URL(BSC_RPC());
    const mod = url.protocol === 'https:' ? https : http;
    let settled = false;
    const timeout = setTimeout(() => { if (!settled) { settled = true; reject(new Error('RPC 8s超时 ' + url.hostname)); rotateRpc(); } }, 8000);
    const req = mod.request({
      hostname: url.hostname, port: 443, path: url.pathname,
      method: 'POST', headers: { 'Content-Type': 'application/json' },
    }, (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        if (settled) return;
        clearTimeout(timeout);
        settled = true;
        try {
          const json = JSON.parse(data);
          if (json.error) reject(new Error(json.error.message));
          else resolve(json.result);
        } catch (e) { reject(e); }
      });
    });
    req.on('error', (err) => { if (!settled) { settled = true; clearTimeout(timeout); rotateRpc(); reject(err); } });
    req.setTimeout(10000, () => { if (!settled) { settled = true; clearTimeout(timeout); req.destroy(); rotateRpc(); reject(new Error('RPC timeout')); } });
    req.write(body);
    req.end();
  });
}

// ═══════════════════════════════════
class UserTrader {
  /**
   * @param {Object} opts
   * @param {Object} opts.userDB      — UserDB 实例 (saas/server.js 里的)
   * @param {Object} opts.dataBus     — DataBus 实例（共享行情数据）
   * @param {Object} opts.aiEngine    — AIDecisionEngine 实例（可选，共享 AI）
   * @param {string} opts.traderKey   — 平台执行器私钥
   * @param {number} opts.intervalMs  — 交易循环间隔（默认 60s）
   */
  constructor(opts = {}) {
    this.userDB = opts.userDB;
    this.dataBus = opts.dataBus;
    this.traderKey = opts.traderKey || process.env.TRADER_PRIVATE_KEY || '';
    if (!this.traderKey) {
      this._log('⚠️ WARNING: TRADER_PRIVATE_KEY 未配置！交易执行将无法工作');
    }
    this.intervalMs = opts.intervalMs || 60000; // 60 秒一轮
 
    this.running = false;
    this._timer = null;
    this._cycleCount = 0;
    this._userStates = {}; // wallet → { lastCycle, positions, pnl, ... }
 
    // v50: 初始化多策略管理器
    this.strategyManager = new StrategyManager({
      grid: { gridSize: 10, gridSpacing: 0.02, positionSize: 0.1 },
      dca: { interval: 24 * 60 * 60 * 1000, amountPerInterval: 50, maxPositions: 5 },
      kelly: { maxPositionPct: 0.25, minPositionPct: 0.05, maxRiskPerTrade: 0.02, kellyFraction: 0.5 },
      volatility: { lookbackPeriod: 20 },
    });
    this._log(`🧠 StrategyManager 已初始化 (网格/DCA/多时间框架/凯利/波动率)`);
 
    // v50: 用户交易统计（用于凯利公式）
    this._userStats = {}; // wallet → { wins, losses, totalPnl, avgWin, avgLoss, winRate }
 
    // 加载持久化状态
    this._loadState();
 
    this._log('UserTrader initialized (v50 多策略引擎)');
  }
 
  _log(msg) {
    console.log(`[UserTrader] ${new Date().toISOString()} ${msg}`);
  }
 
  // ═══════════════════════════════════
  // 启动 / 停止
  // ═══════════════════════════════════
  start() {
    if (this.running) return;
    this.running = true;
    this._log('🚀 UserTrader 启动，间隔 ' + (this.intervalMs / 1000) + 's');
    // 等待 DataBus 加载完成后再开始交易（避免 klines 为空）
    this._waitForDataBus().then(() => this._loop());
  }
 
  async _waitForDataBus(maxWait = 30000) {
    const start = Date.now();
    while (Date.now() - start < maxWait) {
      const klinesCount = Object.keys(this.dataBus?.klines || {}).length;
      if (klinesCount >= 5) {
        this._log(`✅ DataBus 就绪: ${klinesCount} 个交易对有K线`);
        return;
      }
      this._log(`⏳ 等待 DataBus 加载... klines=${klinesCount}`);
      await new Promise(r => setTimeout(r, 2000));
    }
    this._log(`⚠️ DataBus 加载超时，继续执行`);
  }
 
  stop() {
    this.running = false;
    if (this._timer) clearTimeout(this._timer);
    this._log('UserTrader 停止');
  }
 
  async _loop() {
    if (!this.running) return;
    try {
      await this._cycle();
    } catch (e) {
      this._log(`❌ 循环异常: ${e.message}`);
    }
    this._timer = setTimeout(() => this._loop(), this.intervalMs);
  }
 
  // ═══════════════════════════════════
  // 单次交易循环
  // ═══════════════════════════════════
  async _cycle() {
    this._cycleCount++;
    // [audit#24] 动态更新 BNB 价格
    try {
      const md = this.dataBus?.marketData?.['BNBUSDT'];
      if (md?.price) _bnbPriceUsd = md.price;
    } catch(e) {}
    const users = this.userDB.users || {};
    const activeUsers = Object.entries(users).filter(([_, u]) => u.tradingEnabled && u.vaultAddress);
 
    if (activeUsers.length === 0) {
      if (this._cycleCount % 10 === 0) {
        this._log(`第 ${this._cycleCount} 轮: 无活跃用户`);
      }
      return;
    }
 
    this._log(`第 ${this._cycleCount} 轮: ${activeUsers.length} 个活跃用户`);
 
    for (const [wallet, userData] of activeUsers) {
      try {
        const t0 = Date.now();
        await this._tradeForUser(wallet, userData);
        if (Date.now() - t0 > 5000) {
          this._log(`⏱️ 用户 ${wallet.slice(0,10)} 交易耗时 ${((Date.now()-t0)/1000).toFixed(1)}s`);
        }
      } catch (e) {
        this._log(`❌ 用户 ${wallet.slice(0, 10)} 交易失败: ${e.message}`);
      } finally {
        _userLocks.delete(wallet);
      }
    }
 
    this._saveState();
  }
 
  // ═══════════════════════════════════
  // 为单个用户交易
  // ═══════════════════════════════════
  async _tradeForUser(wallet, userData) {
    const vault = userData.vaultAddress;
    if (!vault) { this._log(`[trade] ${wallet.slice(0,10)} 无vault`); return; }
    // 并发锁：上一轮未完成时跳过
    if (_userLocks.get(wallet)) { this._log(`[trade] ${wallet.slice(0,10)} locked, 跳过`); return; }
    _userLocks.set(wallet, true);
 
    // v63修复: 持仓检查优先于余额检查！
    //   之前余额<10直接return, 导致低余额用户的持仓永远无法止损/止盈
    //   现在先检查持仓, 再判断余额是否够开新仓
 
    // 1. 读取用户策略（提前, 因为持仓检查需要 strategyConfig）
    const strategy = userData.strategy || 'balanced';
    const strategyConfig = {
      // v62: DEX高成本适配 — 止盈ATR×5(远超1.6%成本), 止损固定3%, 最低ATR 2.5%
      // v63: 回测优化参数 — 网格搜索最优组合
      // 回测1500根1h K线: ROI=+5.76%, 夏普12.98, 盈亏比2.46
      conservative: { maxPositionPct: 0.15, leverage: 1, slPct: 0.05, tpPct: 0.08, minScore: 7, maxPositions: 2, cooldownMs: 14400000, timeoutHrs: 24, useGrid: false, useDCA: false },
      balanced:     { maxPositionPct: 0.20, leverage: 1, slPct: 0.05, tpPct: 0.08, minScore: 7, maxPositions: 3, cooldownMs: 14400000, timeoutHrs: 20, useGrid: true, useDCA: false },
      aggressive:   { maxPositionPct: 0.25, leverage: 1, slPct: 0.05, tpPct: 0.10, minScore: 6, maxPositions: 3, cooldownMs: 14400000, timeoutHrs: 24, useGrid: true, useDCA: true },
      grid:         { maxPositionPct: 0.15, leverage: 1, slPct: 0.02, tpPct: 0.04, minScore: 3, maxPositions: 5, cooldownMs: 7200000, timeoutHrs: 12, useGrid: true, useDCA: false, gridOnly: true },
      dca:          { maxPositionPct: 0.20, leverage: 1, slPct: 0.05, tpPct: 0.20, minScore: 2, maxPositions: 5, cooldownMs: 86400000, timeoutHrs: 168, useGrid: false, useDCA: true, dcaOnly: true },
    }[strategy] || { maxPositionPct: 0.20, leverage: 1, slPct: 0.03, tpPct: 0.06, minScore: 6, maxPositions: 3, cooldownMs: 14400000, timeoutHrs: 20, useGrid: true, useDCA: false };
 
    // 2. 先检查已有仓位是否需要平仓（v63: 移到余额检查之前！）
    const state = this._getUserState(wallet);
    const currentPositions = Object.keys(state.positions || {});
    if (currentPositions.length > 0) {
      for (const sym of currentPositions) {
        const existingPos = state.positions?.[sym];
        if (!existingPos) continue;
        const md = this.dataBus?.marketData?.[sym];
        const currentPrice = md?.price || md?.lastPrice;
        if (!currentPrice) continue;
        await this._checkClosePosition(wallet, vault, sym, existingPos, currentPrice, strategyConfig, state);
      }
    }
 
    // 3. 余额检查（仅影响开新仓, 不影响持仓监控）
    let usdtBalance = Number(userData.usdtBalance) || 0;
    const userTradeAmount = Number(userData.tradeAmount) || 0;
    if (userTradeAmount <= 0) { this._log(`[trade] ${wallet.slice(0,10)} tradeAmount=0 (持仓${currentPositions.length}个已检查)`); return; }
    usdtBalance = Math.min(usdtBalance, userTradeAmount);
    if (usdtBalance < MIN_TRADE_USDT) { this._log(`[trade] ${wallet.slice(0,10)} balance=${usdtBalance.toFixed(2)} < ${MIN_TRADE_USDT} (持仓${currentPositions.length}个已检查)`); return; }
    if (this._cycleCount % 5 === 0) this._log(`[trade] ${wallet.slice(0,10)} balance=${usdtBalance.toFixed(2)} strategy=${userData.strategy}`);
 
    // 4. 检查仓位是否已满
    const afterClosePositions = Object.keys(state.positions || {});
    const remainingSlots = strategyConfig.maxPositions - afterClosePositions.length;
    if (remainingSlots <= 0) return; // 仓位已满
 
    // 5. 多维度评分筛选
    const topSymbols = this._selectTopSymbols(15);
    if (this._cycleCount % 3 === 0) {
      this._log(`🔍 ${wallet.slice(0, 10)} 评分结果: ${topSymbols.length}个候选, marketData=${Object.keys(this.dataBus?.marketData||{}).length} klines=${Object.keys(this.dataBus?.klines||{}).length}, usdtBalance=${usdtBalance.toFixed(2)}`);
      if (topSymbols.length > 0) {
        this._log(`   Top候选: ${topSymbols.slice(0,3).map(s => `${s.symbol}=${s.score}(${s.side})`).join(', ')}`);
      }
    }
 
    // 持仓检查已在上面余额检查之前完成（v63修复）
    // 重新计算剩余仓位（可能有刚平的）
    const remainingSlots2 = strategyConfig.maxPositions - afterClosePositions.length;
    if (remainingSlots2 <= 0) return;
 
    // v21: 冷却过滤 — 平仓后cooldownMs内不再开同币
    const cooldowns = state.cooldowns || {};
    const now = Date.now();
 
    // 6. 新候选（过滤掉已有仓位+冷却中）
    const candidates = topSymbols
      .filter(s => s.score >= strategyConfig.minScore && !afterClosePositions.includes(s.symbol))
      .filter(s => {
        const lastClose = cooldowns[s.symbol] || 0;
        return (now - lastClose) > strategyConfig.cooldownMs;
      })
      .slice(0, remainingSlots2);
 
    if (candidates.length === 0) {
      if (topSymbols.length > 0 && this._cycleCount % 5 === 0) {
        this._log(`📊 ${wallet.slice(0, 10)} ${topSymbols.length}个候选但分数不足(min=${strategyConfig.minScore}): ${topSymbols.slice(0,3).map(s=>s.symbol+'='+s.score).join(',')}`);
      }
      return;
    }
 
    // 7. 逐个开仓
    for (const cand of candidates) {
      try {
        await this._openPosition(wallet, vault, cand, usdtBalance, strategyConfig, state);
        // 刷新余额（上一笔已花掉一部分）
        try {
          const raw = await erc20Balance(USDT_ADDRESS, vault);
          usdtBalance = Number(raw) / 1e18;
        } catch (e) {}
      } catch (e) {
        this._log(`❌ 开仓异常 ${cand.symbol}: ${e.message}`);
        if (e.stack && this._cycleCount % 5 === 0) this._log(`   stack: ${e.stack.split('\n')[1]?.trim()}`);
      }
    }
  }
 
  // ═══════════════════════════════════
  // v50: 精细开仓 — 凯利公式 + 波动率自适应 + 多策略信号
  // ═══════════════════════════════════
  async _openPosition(wallet, vault, cand, usdtBalance, config, state) {
    const { symbol, score, side, currentPrice, atrPct, atr, confidence, volRegime } = cand;
    this._log(`[v50-open] ${symbol} 进入开仓流程 score=${score} side=${side} price=${currentPrice} atrPct=${atrPct} atr=${atr}`);
    const atrPctValue = atrPct || (atr && currentPrice ? atr / currentPrice : 0.02);
 
    // --- 5a. 盈亏比检查 ---
    const totalCostPct = (PANCAKE_FEE_BPS + SLIPPAGE_BPS) / 10000 * 2 + 0.001; // ~1.5%
    // v50: 多策略确认后降低盈亏比要求（从2%降到0.5%）
    // v62: 止盈目标提升到ATR×5，确保远超1.6% DEX成本
    const minExpectedMove = totalCostPct + 0.01; // 1.6% + 1% = 2.6% 最低预期收益
    const expectedMove = atrPctValue * 5; // v62: 2.5→5, 确保止盈远超成本
    const pctB = cand.pctB || 0.5;
    if (expectedMove < minExpectedMove && pctB > 0.15) {
      if (this._cycleCount % 5 === 0) {
        this._log(`[v50-open] ${symbol} 跳过: expectedMove=${(expectedMove*100).toFixed(2)}% < min=${(minExpectedMove*100).toFixed(2)}% pctB=${pctB.toFixed(3)}`);
      }
      return;
    }
 
    // --- 5b. v50: 凯利公式 + 波动率自适应仓位计算 ---
    const userStats = this._getUserStats(wallet);
    const winRate = userStats.winRate > 0 ? userStats.winRate : 0.5; // 默认50%胜率
    const avgWin = userStats.avgWin > 0 ? userStats.avgWin : 0.03; // 默认3%
    const avgLoss = userStats.avgLoss > 0 ? userStats.avgLoss : 0.01; // 默认1%
 
    // 凯利公式计算最优仓位
    const kellyResult = this.strategyManager.strategies.kelly.calculateKelly(winRate, avgWin, avgLoss);
 
    // 波动率自适应调整
    const volAdjusted = this.strategyManager.strategies.kelly.adjustForMarket(kellyResult.boundedKelly, {
      volatility: atrPctValue,
      trendStrength: Math.abs((cand.ma7 - cand.ma21) / cand.ma21 * 100),
      rsi: cand.rsi || 50,
      correlation: 0,
    });
 
    let kellyPct = volAdjusted.adjustedKelly;
 
    // v50: 凯利仓位下限 — 小账户至少40%，大账户至少10%
    const minKelly = usdtBalance < 100 ? 0.40 : usdtBalance < 500 ? 0.25 : 0.10;
    kellyPct = Math.max(kellyPct, minKelly);
 
    // 根据信号置信度调整（但不低于50%）
    if (confidence) {
      kellyPct *= Math.max(confidence, 0.5);
    }
 
    // 根据评分调整（高分加仓）
    if (score >= 6) kellyPct *= 1.2;
    else if (score >= 4) kellyPct *= 1.0;
    else kellyPct *= 0.8;
 
    // 波动率体制最终调整
    if (volRegime === 'extreme') kellyPct *= 0.5;
    else if (volRegime === 'high') kellyPct *= 0.7;
 
    // 仓位金额
    let positionUsdt = usdtBalance * kellyPct;
    // v50: 小账户允许更高仓位比例
    const effectiveMaxPct = usdtBalance < 100 ? Math.max(config.maxPositionPct, 0.50) : config.maxPositionPct;
    positionUsdt = Math.min(positionUsdt, usdtBalance * effectiveMaxPct);
    positionUsdt = Math.min(positionUsdt, MAX_SINGLE_TRADE_USDT);
 
    // 回撤检查
    const drawdownCheck = this.strategyManager.strategies.kelly.updateDrawdown(usdtBalance);
    if (drawdownCheck.shouldReduce) {
      positionUsdt *= drawdownCheck.reductionFactor;
      this._log(`⚠️ ${wallet.slice(0,10)} 回撤${(drawdownCheck.drawdown*100).toFixed(1)}% > ${(this.strategyManager.strategies.kelly.maxDrawdown*100).toFixed(0)}%, 仓位减半`);
    }
 
    if (positionUsdt < MIN_TRADE_USDT) {
      this._log(`[v50-open] ${symbol} 仓位太小: positionUsdt=$${positionUsdt.toFixed(2)} < MIN=$${MIN_TRADE_USDT} kellyPct=${(kellyPct*100).toFixed(2)}% confidence=${confidence}`);
      return;
    }
 
    // 止损距离
    const atrValue = atr || (currentPrice * 0.02);
    // v63: 止损5%（回测最优），止损距离取max(5%, 3×ATR)
    const stopDistPct = Math.max(config.slPct || 0.05, atrValue ? (3 * atrValue / currentPrice) : config.slPct);
 
    // 预期利润检查
    const estProfit = positionUsdt * (expectedMove - totalCostPct) - GAS_COST_USD();
    if (estProfit < 1) {
      if (this._cycleCount % 5 === 0) {
        this._log(`[v50-open] ${symbol} 跳过: estProfit=$${estProfit.toFixed(2)} positionUsdt=$${positionUsdt.toFixed(2)} expectedMove=${(expectedMove*100).toFixed(2)}% costPct=${(totalCostPct*100).toFixed(2)}%`);
      }
      return;
    }
 
    // --- 5c. 滑点保护 ---
    const slippageMultiplier = positionUsdt > 500 ? 1.5 : positionUsdt > 200 ? 1.2 : 1;
    const effectiveSlippageBps = Math.floor(SLIPPAGE_BPS * slippageMultiplier);
 
    // --- 5d. swap 路径 ---
    const tokenIn = USDT_ADDRESS;
    const tokenOut = this._getBscToken(symbol);
    if (!tokenOut) return;
 
    this._log(
      `📈 ${wallet.slice(0, 8)} 开仓 ${symbol} ${side}`
      + ` | 金额 $${positionUsdt.toFixed(2)}`
      + ` | 凯利仓位 ${(kellyPct * 100).toFixed(1)}%`
      + ` | 分数 ${score}`
      + ` | 置信度 ${(confidence || 0).toFixed(2)}`
      + ` | 波动率 ${volRegime || 'medium'}`
      + ` | ATR ${(atrPctValue * 100).toFixed(2)}%`
      + ` | 预期 ${(expectedMove * 100).toFixed(2)}%`
      + ` | 成本 ${(totalCostPct * 100).toFixed(2)}%`
    );
 
    // --- 5e. 执行 swap ---
    const receipt = await this._executeSwapInVault(
      vault, tokenIn, tokenOut, positionUsdt, effectiveSlippageBps
    );
 
    // --- 5f. 记录仓位（含精细数据）---
    if (!state.positions) state.positions = {};
    state.positions[symbol] = {
      side, entryPrice: currentPrice,
        atrPct: atrPctValue || 2,
      amount: positionUsdt, openTime: Date.now(),
      sl: currentPrice * (1 - stopDistPct),
      tp: currentPrice * (1 + expectedMove),
      atr, atrPct, stopDistPct, score,
      costPct: totalCostPct,
      slippageBps: effectiveSlippageBps,
    };
 
    this._logTrade(wallet, vault, {
      symbol, action: side, amount: positionUsdt,
      price: currentPrice, timestamp: Date.now(),
      score, atrPct, stopDistPct, expectedMove,
    });
  }
 
  // ═══════════════════════════════════
  // 精细平仓（移动止损/趋势评分/持仓时间/盈利保护）
  // ═══════════════════════════════════
  async _checkClosePosition(wallet, vault, symbol, pos, currentPrice, config, state) {
    const entryPrice = pos.entryPrice;
    if (!entryPrice || !currentPrice) return;
 
    const holdTime = Date.now() - pos.openTime;
    const holdMinutes = holdTime / 60000;
    const holdHours = holdTime / 3600000;
 
    // --- 实际 PnL ---
    let rawPnlPct;
    if (!currentPrice || !entryPrice || !isFinite(currentPrice) || !isFinite(entryPrice)) {
      rawPnlPct = 0; // v67-fix: 防止 NaN
    } else if (pos.side === 'LONG') {
      rawPnlPct = (currentPrice - entryPrice) / entryPrice;
    } else {
      rawPnlPct = (entryPrice - currentPrice) / entryPrice;
    }
    // costPct 是双边费用（开+平），但当前还没平仓，只扣单边
    const singleSideCost = (pos.costPct || ((PANCAKE_FEE_BPS + SLIPPAGE_BPS) / 10000 * 2 + 0.001)) / 2;
    const netPnlPct = rawPnlPct - singleSideCost;
    const leveragedPnl = netPnlPct * (pos.leverage || 2);
 
    // 记录峰值（如果从旧策略迁移，重置不合理的峰值）
    if (!pos._peakPnl) pos._peakPnl = 0;
    if (pos._peakPnl > 0 && pos._peakPnl < 0.03) pos._peakPnl = 0; // 旧策略遗留清理
    if (leveragedPnl > pos._peakPnl) pos._peakPnl = leveragedPnl;
 
    let shouldClose = false;
    let reason = '';
 
    // ═══ v21: 基于费用的精确止盈止损 ═══
    // 双边费用约 1.5%，所以：
    //   止损 = -1%（净亏2.5%含费用）
    //   止盈 = +3%（净赚1.5%含费用）
    // ═══ v60: 动态ATR止损 + 15分钟保护期 ═══
    // v60修复：之前硬止损-1%导致开仓0.0h就止损，策略没时间验证
    // 新逻辑：
    //   - 保护期15分钟内不检查止损（除非极端亏损-5%）
    //   - 止损线 = ATR的2倍（动态适应波动率），限制2%-5%
    //   - 极端亏损-5%保护期内也触发
    // v63: 回测优化 — 止损5%（网格搜索最优）
    const dynamicStopPct = config.slPct || 0.05;
 
    // 1. 极端亏损保护（保护期内也触发）
    if (netPnlPct <= -0.05) {
      shouldClose = true;
      reason = `极端止损 净亏 ${(netPnlPct * 100).toFixed(2)}%`;
    }
    // 2. 保护期15分钟内不检查其他止损
    else if (holdMinutes < 15) { // v63: 30→15分钟，回测最优
      return;
    }
    // 3. ATR动态止损
    else if (netPnlPct <= -dynamicStopPct) {
      shouldClose = true;
      reason = `ATR止损 净亏 ${(netPnlPct * 100).toFixed(2)}% (线${(dynamicStopPct * 100).toFixed(1)}%)`;
    }
    // 4. 价格止损
    else if (pos.sl && currentPrice <= pos.sl) {
      shouldClose = true;
      reason = `价格止损 ${currentPrice.toFixed(4)} <= ${pos.sl.toFixed(4)}`;
    }
    // 5. 止盈：净赚超过+3%
    else if (netPnlPct >= 0.05) { // v62: 3%→5%，扣1.6%成本后净赚3.4%
      shouldClose = true;
      reason = `止盈 净赚 ${(netPnlPct * 100).toFixed(1)}%（含费用）`;
    }

      amountIn, [fixAddr(tokenIn), fixAddr(tokenOut)]

      throw new Error('QUOTE_RPC_FAILED: ' + e.message?.slice(0, 100));

      throw new Error('QUOTE_EMPTY: ' + String(quoteRaw).slice(0, 50));

      quoteResult = coder.decodeFunctionResult('getAmountsOut', quoteRaw);
    } catch(e) {
      throw new Error('QUOTE_DECODE_FAILED: ' + e.message?.slice(0, 100));
    }
    if (!quoteResult || !quoteResult[0] || typeof quoteResult[0][1] === 'undefined') {
      throw new Error('QUOTE_INVALID: no amounts array');

    const minOut = quoteResult[0][1] * BigInt(10000 - slippageBps) / 10000n;

      'function executeSwap(address dex, address tokenIn, address tokenOut, uint256 amountIn, uint256 minAmountOut) external returns (uint256)'

      throw new Error('VAULT_CALL_FAILED: ' + (e.reason || e.message?.slice(0, 150)));

    const receipt = await this._waitForReceipt(wallet.provider, tx.hash);

  }

  _sma(data, period) {
    if (data.length < period) return null;
    const slice = data.slice(-period);
    return slice.reduce((a, b) => a + b, 0) / period;
  }
 
  _ema(data, period) {
    if (data.length < period) return null;
    const k = 2 / (period + 1);
    let ema = data.slice(0, period).reduce((a, b) => a + b, 0) / period;
    for (let i = period; i < data.length; i++) {
      ema = data[i] * k + ema * (1 - k);
    }
    return ema;
  }
 
  _rsi(closes, period = 14) {
    if (closes.length < period + 1) return 50;
    let gains = 0, losses = 0;
    for (let i = closes.length - period; i < closes.length; i++) {
      const diff = closes[i] - closes[i - 1];
      if (diff > 0) gains += diff; else losses -= diff;
    }
    if (losses === 0) return 100;
    const rs = gains / losses;
    return 100 - (100 / (1 + rs));
  }
 
  _atr(klines, period = 14) {
    if (klines.length < period + 1) return null;
    const trs = [];
    for (let i = klines.length - period; i < klines.length; i++) {
      const h = klines[i].high ?? klines[i][2] ?? 0;
      const l = klines[i].low ?? klines[i][3] ?? 0;
      const prevC = klines[i - 1].close ?? klines[i - 1][4] ?? 0;
      const tr = Math.max(h - l, Math.abs(h - prevC), Math.abs(l - prevC));
      trs.push(tr);
    }
    return trs.reduce((a, b) => a + b, 0) / trs.length;
  }

  _selectTopSymbols(maxCount = 10) {

    const scored = [];
 
    // 1. 先评 DataBus 已有的交易对（使用正确的 klines 引用）

        if (this._cycleCount % 10 === 0) this._log(`[selectTop] ${sym} SKIP: klines=${klines?.length||0}`);
        continue;
      }
      const result = this._scoreSymbol(sym);
      if (result.score >= 3) {
        scored.push({ symbol: sym, ...result });
      } else if (this._cycleCount % 3 === 0) {
        this._log(`[selectTop] ${sym} score=${result.score} side=${result.side} klines=${klines.length}`);
      }
    }
 
    // 不再动态拉取新交易对（HTTP请求太慢会卡住cycle，且WS不稳定导致数据不完整）

    // 更多交易对的评分等WS稳定后由管理员引擎的Scanner负责补充
 
    scored.sort((a, b) => b.score - a.score);
    return scored.slice(0, maxCount);

      const candidates = tickers

          && parseFloat(t.quoteVolume || 0) > 50000000)
        .sort((a, b) => Math.abs(parseFloat(b.priceChangePercent)) - Math.abs(parseFloat(a.priceChangePercent)))

  // [audit#6] 轮询等待交易确认（避免 ethers v6 url.clone bug）
  async _waitForReceipt(provider, txHash, maxWaitMs = 120000) {
    const start = Date.now();
    let errorCount = 0;
    while (Date.now() - start < maxWaitMs) {
      await new Promise(r => setTimeout(r, 3000));
      try {
        const receipt = await provider.getTransactionReceipt(txHash);
        if (receipt) {
          this._log(`✅ tx confirmed: ${txHash?.slice(0,18)}... status=${receipt.status} gas=${receipt.gasUsed}`);
          return receipt;
        }
        errorCount = 0; // 重置错误计数
      } catch (e) {
        errorCount++;
        if (errorCount % 10 === 0) this._log(`⏳ tx确认重试 ${errorCount}次: ${e.message?.slice(0,60)}`);
        // 连续20次错误则放弃
        if (errorCount >= 20) throw new Error(`tx确认失败(20次重试): ${e.message?.slice(0,80)}`);
      }
    }
    throw new Error(`Transaction timeout: ${txHash?.slice(0,18)} not confirmed within ${maxWaitMs/1000}s`);
  }

  // v50: 更新用户统计（平仓时调用）
  _updateUserStats(wallet, pnlPct, pnlUsdt) {
    // v67-fix: 防止 NaN 污染统计
    if (typeof pnlPct !== 'number' || !isFinite(pnlPct)) pnlPct = 0;
    if (typeof pnlUsdt !== 'number' || !isFinite(pnlUsdt)) pnlUsdt = 0;
 
    const stats = this._getUserStats(wallet);
    stats.totalPnl += pnlUsdt;
    if (pnlPct > 0) {
      stats.wins++;
      stats.avgWin = (stats.avgWin * (stats.wins - 1) + pnlPct) / stats.wins;
    } else if (pnlPct < 0) {
      stats.losses++;
      stats.avgLoss = (stats.avgLoss * (stats.losses - 1) + Math.abs(pnlPct)) / stats.losses;
    }
    // pnlPct === 0: 不计入胜负，避免 avgLoss 被稀释到 0
    const total = stats.wins + stats.losses;
    stats.winRate = total > 0 ? stats.wins / total : 0;
  }
